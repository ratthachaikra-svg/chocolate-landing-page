4. โค้ดสำหรับนำไปวางใน Google Apps Script

คัดลอกโค้ดนี้ไปวางใน Extensions > Apps Script ของ Google Sheet แล้ว Deploy เป็น Web app

const SHEET_NAME = 'Orders';

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  const headers = [
    'Timestamp',
    'Order ID',
    'Customer Name',
    'Phone',
    'Address',
    'Shipping Method',
    'Items Summary',
    'Total Qty',
    'Grand Total',
    'Note',
    'Pack Status'
  ];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(headers);
  }
  return sheet;
}

function doPost(e) {
  try {
    const sheet = getOrCreateSheet();
    const data = JSON.parse(e.postData.contents);

    if (data.action === 'updateStatus') {
      const values = sheet.getDataRange().getValues();
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][1]) === String(data.orderId)) {
          sheet.getRange(i + 1, 11).setValue(data.packStatus || 'PACKED');
          return jsonResponse({ ok: true, updated: data.orderId });
        }
      }
      return jsonResponse({ ok: false, error: 'Order not found' });
    }

    const row = [
      data.timestamp || new Date().toISOString(),
      data.orderId || '',
      data.customerName || '',
      data.phone || '',
      data.address || '',
      data.shippingMethod || '',
      data.itemsSummary || '',
      data.totalQty || 0,
      data.grandTotal || 0,
      data.note || '',
      data.packStatus || 'PENDING'
    ];

    sheet.appendRow(row);
    return jsonResponse({ ok: true, orderId: data.orderId });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message });
  }
}

function doGet() {
  try {
    const sheet = getOrCreateSheet();
    const rows = sheet.getDataRange().getValues();
    if (rows.length <= 1) {
      return jsonResponse({ ok: true, orders: [] });
    }

    const orders = [];
    for (let i = rows.length - 1; i >= 1; i--) {
      const r = rows[i];
      if (!r[1]) continue;
      orders.push({
        timestamp: r[0],
        orderId: String(r[1]),
        customerName: String(r[2] || ''),
        phone: String(r[3] || ''),
        address: String(r[4] || ''),
        shippingMethod: String(r[5] || ''),
        itemsSummary: String(r[6] || ''),
        totalQty: Number(r[7] || 0),
        grandTotal: Number(r[8] || 0),
        note: String(r[9] || ''),
        packStatus: String(r[10] || 'PENDING')
      });
    }

    return jsonResponse({ ok: true, orders: orders });
  } catch (err) {
    return jsonResponse({ ok: false, error: err.message });
  }
}

function jsonResponse(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
